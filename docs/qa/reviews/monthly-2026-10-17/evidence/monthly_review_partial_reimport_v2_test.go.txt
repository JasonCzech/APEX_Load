package api

import (
 "errors"
 "net/http"
 "path/filepath"
 "testing"

 "github.com/jasonczech/apexload/pkg/dsl"
 "github.com/jasonczech/apexload/pkg/storage"
)

type monthlyReviewPartialReimportStore struct {
 *storage.SQLiteStore
 armed bool
 updates int
}
func(s *monthlyReviewPartialReimportStore)UpdateScriptAuthoringIfCurrent(script *dsl.Script,expected storage.ScriptAuthoringPrecondition)(*storage.ScriptAuthoringMutationResult,error){
 if s.armed {s.updates++;if s.updates>1{return nil,errors.New("transient storage outage after first update")}}
 return s.SQLiteStore.UpdateScriptAuthoringIfCurrent(script,expected)
}

func TestMonthlyReviewPartialReimportCanResumeAfterStorageRecovers(t *testing.T){
 base,err:=storage.NewSQLiteStore(filepath.Join(t.TempDir(),"partial.db"));if err!=nil{t.Fatal(err)};t.Cleanup(func(){base.Close()})
 store:=&monthlyReviewPartialReimportStore{SQLiteStore:base}
 server:=NewServer(ServerConfig{Store:store})
 first:=asyncAPIReimportScripts(t,postAsyncAPIBatch(t,server,map[string]any{"operation_id":"monthly-partial-first","scripts":[]any{asyncAPIReimportScript("Orders","publish v1"),asyncAPIReimportScript("Refunds","publish v1")}}))
 var targets []any
 for _,item:=range first{current,_:=server.runner.GetScript(item.ID);digest,e:=importedScriptContentDigest(current.Script);if e!=nil{t.Fatal(e)};targets=append(targets,map[string]any{"script_id":item.ID,"expected_updated_at":current.Script.UpdatedAt,"expected_content_digest":digest})}
 request:=map[string]any{"operation_id":"monthly-partial-second","scripts":[]any{asyncAPIReimportScript("Orders","publish v2"),asyncAPIReimportScript("Refunds","publish v2")},"update_targets":targets}
 store.armed=true
 failed:=postAsyncAPIBatch(t,server,request)
 if failed.Code!=http.StatusInternalServerError{t.Fatalf("first attempt status %d: %s",failed.Code,failed.Body.String())}
 one,_:=server.runner.GetScript(first[0].ID);two,_:=server.runner.GetScript(first[1].ID)
 if one.Script.Actions[0].Name!="publish v2"||two.Script.Actions[0].Name!="publish v1"{t.Fatalf("fixture did not reach partial applied state: %s / %s",one.Script.Actions[0].Name,two.Script.Actions[0].Name)}
 store.armed=false
 retry:=postAsyncAPIBatch(t,server,request)
 if retry.Code!=http.StatusCreated&&retry.Code!=http.StatusOK{t.Fatalf("same-operation retry after storage recovered rejected its own partially applied update: %d %s",retry.Code,retry.Body.String())}
}
