package api

import (
 "testing"
 "time"
)

func TestMonthlyReviewPendingPlanStopSurvivesControllerRestart(t *testing.T) {
 store, original, _, plan, firstID := newSequentialRecoveryFixture(t, "skip")
 const runID = "exec-plan-run-recovery"
 // This is the same missing-memory state exercised by the existing
 // TestStopBeforeRecoveryRestoresTheRunIsAppliedOnRestore. It also occurs
 // when recovery discards a provisional state after a persistence failure.
 original.seqMu.Lock()
 delete(original.seqPlanRuns, runID)
 delete(original.seqPlanByTest, firstID)
 original.seqMu.Unlock()
 if _, err := original.StopExecutionPlanRun(plan.ID, runID, testCreationActor{DisplayName:"Alice"}); err != nil {
  t.Fatalf("stop must have been accepted before restart: %v",err)
 }
 // Graceful stopping of the current child can end COMPLETE. The plan-level
 // stop, independent of that outcome, must still suppress the next entry.
 end:=time.Now().UTC()
 if err:=store.UpdateTestStatus(firstID,"COMPLETE",&end,"");err!=nil{t.Fatal(err)}
 original.beginShutdown()
 recovered, backend := acquireRecoveredSequentialRunner(t, store)
 defer recovered.beginShutdown()
 if next, started := waitForStart(t,backend,2*time.Second);started {
  t.Fatalf("controller restart lost accepted Stop plan and launched next entry: %s",next)
 }
}
