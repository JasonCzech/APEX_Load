type monthlyReviewDraftDriver struct { base sqlite.Driver; after func() }
func (d *monthlyReviewDraftDriver) Open(name string) (driver.Conn,error) { c,e:=d.base.Open(name); if e!=nil{return nil,e}; return &monthlyReviewDraftConn{Conn:c,d:d},nil }
type monthlyReviewDraftConn struct { driver.Conn; d *monthlyReviewDraftDriver }
func (c *monthlyReviewDraftConn) ExecContext(ctx context.Context,q string,args []driver.NamedValue)(driver.Result,error){
 r,e:=c.Conn.(driver.ExecerContext).ExecContext(ctx,q,args)
 if e==nil && strings.HasPrefix(q,"UPDATE scenario_drafts SET scenario_id") && c.d.after!=nil { c.d.after() }
 return r,e
}
func (c *monthlyReviewDraftConn) QueryContext(ctx context.Context,q string,args []driver.NamedValue)(driver.Rows,error){ return c.Conn.(driver.QueryerContext).QueryContext(ctx,q,args) }

func TestMonthlyReviewDraftSaveDoesNotAdoptAnotherWritersRevision(t *testing.T) {
 path:=filepath.Join(t.TempDir(),"draft-race.db")
 original,err:=NewSQLiteStore(path); if err!=nil{t.Fatal(err)}
 defer original.Close()
 draft:=&ScenarioDraft{ID:"draft-race",ProjectID:DefaultProjectID,Name:"Initial",Document:`{"form":"initial"}`}
 if err:=original.CreateScenarioDraft(draft);err!=nil{t.Fatal(err)}
 d:=&monthlyReviewDraftDriver{}
 sql.Register("monthly-review-draft-race",d)
 db,err:=sql.Open("monthly-review-draft-race",sqliteDSN(path));if err!=nil{t.Fatal(err)}
 defer db.Close()
 store:=&SQLiteStore{db:&storeDB{db:db,dialect:storeDialectSQLite}}
 var once sync.Once
 d.after=func(){once.Do(func(){
  // Writer B reads the committed A update and saves revision 2 before A's
  // response SELECT, exactly as another HTTP request can do.
  b,e:=original.GetScenarioDraft(DefaultProjectID,draft.ID);if e!=nil{t.Fatal(e)}
  if b.Revision!=2{t.Fatalf("B sees revision %d",b.Revision)}
  b.Name="Writer B";b.Document=`{"form":"B"}`;b.UpdatedBy="B"
  ok,e:=original.UpdateScenarioDraftIfRevision(b,2);if e!=nil||!ok{t.Fatalf("B update: %v %v",ok,e)}
 })}
 a:=&ScenarioDraft{ID:draft.ID,ProjectID:DefaultProjectID,Name:"Writer A",Document:`{"form":"A"}`,UpdatedBy:"A"}
 ok,err:=store.UpdateScenarioDraftIfRevision(a,1);if err!=nil||!ok{t.Fatalf("A update: %v %v",ok,err)}
 if a.Revision!=2 || a.Document!=`{"form":"A"}` { t.Fatalf("A save returned another writer's result: revision=%d document=%s name=%s; expected revision=2 document=A",a.Revision,a.Document,a.Name) }
}
