package engine

import (
	"encoding/json"
	"fmt"
	"testing"

	"github.com/jasonczech/apexload/pkg/convert/paramize"
	"github.com/jasonczech/apexload/pkg/dsl"
)

func TestMonthlyReviewRepeatedGeneratedIDPreservesRequestEquality(t *testing.T) {
	script := &dsl.Script{Name: "generated IDs", Version: "1.0", Protocol: "http"}
	for i, id := range []string{"0a1b2c3d-1111-4aaa-8bbb-000000000001", "0a1b2c3d-2222-4aaa-8bbb-000000000002"} {
		script.Actions = append(script.Actions, dsl.Action{Name: fmt.Sprint(i), Type: "http_request", Method: "POST", Target: "https://example.invalid", Body: fmt.Sprintf(`{"requestId":%q,"parentRequestId":%q}`, id, id)})
	}
	result := paramize.Suggest(script, paramize.Options{EnableGenerated: true})
	selected := map[string]bool{}
	for _, suggestion := range result.Suggestions { selected[suggestion.ID] = true }
	applied, _, err := paramize.ApplySuggestions(script, result.Suggestions, selected)
	if err != nil { t.Fatal(err) }
	if len(applied.Parameters) == 0 { t.Fatal("no generated parameters") }
	vu := &VUser{Script: applied, Variables: map[string]string{}}
	vu.Parameters = NewParameterRuntime(applied.Parameters, nil, 0)
	vu.Parameters.Prime()
	resolved := vu.resolveAction(applied.Actions[0])
	var body map[string]string
	if err := json.Unmarshal([]byte(resolved.Body), &body); err != nil { t.Fatal(err) }
	if body["requestId"] != body["parentRequestId"] {
		t.Fatalf("same recorded ID diverged after generation: body=%s parameters=%+v", resolved.Body, applied.Parameters)
	}
}
