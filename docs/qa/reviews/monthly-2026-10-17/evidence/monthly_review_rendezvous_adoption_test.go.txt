package api

import (
 "context"
 "testing"
 "time"

 "github.com/jasonczech/apexload/pkg/distributed/generatorrpc"
 "github.com/jasonczech/apexload/pkg/distributed/rendezvous"
 "github.com/jasonczech/apexload/pkg/storage"
)

type monthlyReviewRendezvousResumeClient struct {
 generatorClient
 stream *generatorrpc.MetricsStream
 handler rendezvous.Client
}
func (c *monthlyReviewRendezvousResumeClient) OpenRendezvous(_ context.Context,_,_ string,_ uint64,_ []byte,h rendezvous.Client) error { c.handler=h;return nil }
func (c *monthlyReviewRendezvousResumeClient) OpenMetricsStreamForTestGeneration(context.Context,string,uint64)(*generatorrpc.MetricsStream,error) {return c.stream,nil}
func (c *monthlyReviewRendezvousResumeClient) Close() error{return nil}

func TestMonthlyReviewAdoptedRendezvousRemainsServing(t *testing.T){
 runner,store,_:=newAdoptionTestRunner(t)
 term,err:=runner.captureExecutionLeaseTerm();if err!=nil{t.Fatal(err)}
 group:=rendezvousTestGroup()
 started:=time.Now().UTC()
 runCtx,cancel:=context.WithCancel(context.Background())
 run:=&ManagedTest{ID:"review-adopt-rendezvous",Name:"review",ProjectID:storage.DefaultProjectID,Status:"RUNNING",Groups:[]TestGroup{group},StartedAt:&started,Duration:time.Minute,executionCtx:runCtx,cancel:cancel,executionGeneration:term.generation}
 if err:=store.SaveTest(&storage.TestRecord{ID:run.ID,ProjectID:run.ProjectID,Name:run.Name,Status:run.Status,StartTime:started,Groups:[]any{}});err!=nil{t.Fatal(err)}
 shards:=[]GeneratorShardSummary{{Group:group.Name,Target:"gen-a:9091",VUsers:4}}
 run.remoteGroupShards=map[string][]GeneratorShardSummary{group.Name:shards}
 runner.mu.Lock();runner.tests[run.ID]=run;runner.mu.Unlock()
 t.Cleanup(func(){runner.mu.Lock();run.Status="COMPLETE";cancel();if run.remoteLifecycleCancel!=nil{run.remoteLifecycleCancel()};runner.mu.Unlock()})
 // Model the new leader's Phase A recovery using the real durable coordinator.
 if err:=runner.prepareGroupRendezvous(context.Background(),run,group,shards,false,term);err!=nil{t.Fatal(err)}
 _,stream:=openScriptedMetricsStream(t,run.ID,term.generation)
 client:=&monthlyReviewRendezvousResumeClient{stream:stream}
 run.adoptedResume=&adoptedRunResume{generation:term.generation,ownerID:term.ownerID,secured:true,adopted:map[string]generatorClient{"gen-a:9091":client}}
 runner.resumeAdoptedRun(run)
 runner.mu.RLock();active:=run.Status=="RUNNING"&&run.remoteActivated;runner.mu.RUnlock()
 if !active||client.handler==nil{t.Fatalf("resume failed before serving: active=%v handler=%v",active,client.handler!=nil)}
 _,err=client.handler.Exchange(context.Background(),rendezvous.Batch{Key:"checkout",Round:1,Arrivals:[]uint64{0}})
 if err!=nil {t.Fatalf("successful adoption closed its rendezvous while run is active: %v (runtime present=%v)",err,runner.rendezvousRuntime(run.ID,group.Name)!=nil)}
}
