//go:build analytics_duckdb && cgo

package duckdb

import (
	"context"
	"testing"
	"time"

	"github.com/jasonczech/apexload/pkg/analysis"
)

func TestMonthlyReviewHistogramWindowBoundaries(t *testing.T) {
	cases := []struct {
		name, fixture string
		duration time.Duration
		bucket int
	}{
		{"overlapping count windows", "SELECT * FROM (VALUES (1700000040, 60.0, 0.0), (1700000070, 60.0, 0.0)) t(ts, n, overflow)", 120*time.Second, 120},
		{"histogram extends past query end", "SELECT 1700000040 AS ts, 60.0 AS n, 0.0 AS overflow", 30*time.Second, 60},
	}
	for _, tc := range cases { t.Run(tc.name, func(t *testing.T) {
		e, path := nativeFixture(t, tc.fixture)
		mapping := Mapping{Version:1, Timestamp:Column{"ts"}, TimestampEncoding:"unix_s", Timezone:"UTC", Measures:[]Measure{{
			Value:Column{"overflow"}, Name:"requests", Kind:analysis.KindHistogram, Unit:"ms",
			Histogram:&analysis.ParquetHistogram{Bounds:[]string{"100", "+Inf"}, BucketCounts:analysis.HistogramPerBin, Temporality:analysis.HistogramDelta, WindowSeconds:60, CountColumns:[]Column{{"n"}}},
		}}}
		result := normalizeSummaryFixture(t, e, path, mapping, false)
		start := summaryStart()
		points, err := e.Series(context.Background(), result.Files, SeriesQuery{ProjectID:"p", DatasetID:"d", Metric:"requests", From:start, To:start.Add(tc.duration), BucketSeconds:tc.bucket, Aggregation:"count"})
		if err != nil { t.Fatal(err) }
		if len(points)==0 { t.Fatal("no points") }
		for _, p := range points { if p.Quality == analysis.QualityValid { t.Errorf("inexact histogram reported as valid: value=%v timestamp=%s", *p.Value, p.TimestampUTC) } }
	}) }
}
