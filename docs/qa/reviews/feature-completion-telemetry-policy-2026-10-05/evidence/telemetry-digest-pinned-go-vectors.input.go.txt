 "crypto/sha256"
 "encoding/hex"
 "encoding/json"
 "fmt"
 "math"
 "time"
 "github.com/jasonczech/apexload/pkg/api"
)
func main(){
 stages:=[]string{"sample_age","generator_processing","transport","controller_fanout","browser_render"}
 for _,n:= range []float64{11,17,999,0,math.Copysign(0,-1),0.1,1e-12,1e-7,1e-6,1e20,1e21,math.SmallestNonzeroFloat64,math.MaxFloat64,1.0000000000000002,9007199254740992,1000000000000000100} {
 p:=api.TelemetryBudgetPolicy{SchemaVersion:"apexload.telemetry_policy.v1",PolicyID:"system-telemetry-budgets",Revision:1,CaptureBoundary:"execution-admission-before-staging",ClassificationAlgorithm:"legacy-mixed-diagnostic.v1",ExpectedControllerPublicationIntervalMs:3000,CadenceScope:"local-controller-ticker",SustainedBreachWindowMs:15000,CoverageRule:"retained-classifications-only;gaps-unknown"}
 for i,v:=range []float64{n,0,750,250,1e-12}{p.Budgets=append(p.Budgets,api.TelemetryStageBudget{Stage:stages[i],Enabled:v>0,LimitMs:v})}
 b,err:=json.Marshal(p);if err!=nil{panic(err)};sum:=sha256.Sum256(b);p.Hash=hex.EncodeToString(sum[:]);p.CapturedAt=time.Date(2026,10,5,12,0,0,0,time.UTC);if !api.ValidateTelemetryBudgetPolicy(&p){panic("actual validator rejected vector")};encoded,_:=json.Marshal(n);fmt.Printf("%s %s\n",encoded,p.Hash)
 }
}
