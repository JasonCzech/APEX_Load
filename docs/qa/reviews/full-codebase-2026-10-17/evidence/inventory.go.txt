package main
import("encoding/json";"go/ast";"go/parser";"go/token";"os";"os/exec";"path/filepath";"sort";"strconv";"strings")
type File struct {Path string `json:"path"`; Lines int `json:"lines"`; Test bool `json:"test"`; Package string `json:"package"`; Imports []string `json:"imports"`}
type Function struct {Path string `json:"path"`; Name string `json:"name"`; Line int `json:"line"`; Lines int `json:"lines"`; Statements int `json:"statements"`}
type Report struct {Files []File `json:"files"`; Functions []Function `json:"functions"`; Excluded []string `json:"excluded"`}
func main(){root:=os.Args[1];os.Chdir(root); raw,err:=exec.Command("git","ls-files","-z").Output();if err!=nil{panic(err)};r:=Report{}
 for _,p:=range strings.Split(string(raw),"\x00"){if p==""{continue};ext:=filepath.Ext(p);if ext!=".go"&&ext!=".ts"&&ext!=".tsx"&&ext!=".py"&&ext!=".js"&&ext!=".mjs"&&ext!=".sh"&&ext!=".tf"&&ext!=".yaml"&&ext!=".yml"{continue};b,err:=os.ReadFile(p);if err!=nil{continue};s:=string(b);head:=s;if len(head)>2048{head=head[:2048]};exclude:=strings.HasPrefix(p,"_retired/")||strings.HasPrefix(p,"third_party/")||strings.HasPrefix(p,"pkg/api/static/")||strings.HasPrefix(p,"dashboard/src/types/generated/")||strings.HasPrefix(p,"docs/")||strings.HasSuffix(p,"/apidocs/swagger.yaml")||strings.Contains(head,"Code generated")||strings.Contains(head,"This file was auto-generated")||strings.Contains(head,"This file is generated")
 if exclude{r.Excluded=append(r.Excluded,p);continue};test:=strings.HasSuffix(p,"_test.go")||strings.Contains(p,".test.")||strings.Contains(p,".spec.")||strings.Contains(p,"/tests/")||strings.Contains(p,"/e2e/");f:=File{Path:p,Lines:strings.Count(s,"\n"),Test:test,Package:filepath.Dir(p)}
 if ext==".go"{fs:=token.NewFileSet();tree,err:=parser.ParseFile(fs,p,b,parser.SkipObjectResolution);if err!=nil{panic(err)};for _,im:=range tree.Imports{ip,_:=strconv.Unquote(im.Path.Value);if strings.HasPrefix(ip,"github.com/jasonczech/apexload/"){f.Imports=append(f.Imports,strings.TrimPrefix(ip,"github.com/jasonczech/apexload/"))}};if !test{for _,d:=range tree.Decls{fn,ok:=d.(*ast.FuncDecl);if !ok||fn.Body==nil{continue};name:=fn.Name.Name;if fn.Recv!=nil&&len(fn.Recv.List)>0{switch x:=fn.Recv.List[0].Type.(type){case *ast.Ident:name=x.Name+"."+name;case *ast.StarExpr:if x,ok:=x.X.(*ast.Ident);ok{name=x.Name+"."+name}}};stmts:=0;ast.Inspect(fn.Body,func(n ast.Node)bool{if _,ok:=n.(ast.Stmt);ok{stmts++};return true});r.Functions=append(r.Functions,Function{p,name,fs.Position(fn.Pos()).Line,fs.Position(fn.End()).Line-fs.Position(fn.Pos()).Line+1,stmts})}}};r.Files=append(r.Files,f)
 };sort.Slice(r.Functions,func(i,j int)bool{return r.Functions[i].Lines>r.Functions[j].Lines});b,_:=json.MarshalIndent(r,"","  ");os.WriteFile(os.Args[2],b,0644)
}
