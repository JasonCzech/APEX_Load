package jsonextract
import "testing"
func TestReviewJSONPathExactInteger(t *testing.T) {
 got,err := JSONPath(`$.id`,[]byte(`{"id":9007199254740993}`))
 if err != nil || got != "9007199254740993" { t.Fatalf("exact integer: got=%q err=%v; want 9007199254740993",got,err) }
}
func TestReviewJSONPathStringIntegerControl(t *testing.T) {
 got,err := JSONPath(`$.id`,[]byte(`{"id":"9007199254740993"}`))
 if err != nil || got != "9007199254740993" { t.Fatalf("got=%q err=%v",got,err) }
}
