func TestFullReviewStatusSLAEvaluation(t *testing.T) {
    for _, tc := range []struct{name, format string; evaluated bool}{
        {"unevaluated_text", "text", false},
        {"unevaluated_json", "json", false},
        {"evaluated_text_control", "text", true},
    } {
        t.Run(tc.name, func(t *testing.T) {
            srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.URL.Path != "/api/v1/tests/probe/sla-status" { t.Errorf("unexpected path %s", r.URL.Path) }
                w.Header().Set("Content-Type", "application/json")
                fmt.Fprintf(w, `{"test_id":"probe","status":"COMPLETE","sla_evaluated":%t,"sla_breached":false}`, tc.evaluated)
            }))
            defer srv.Close()
            previousArgs, previousStdout := os.Args, os.Stdout
            outputPath := filepath.Join(t.TempDir(), "stdout.txt")
            output, err := os.Create(outputPath)
            if err != nil { t.Fatal(err) }
            defer output.Close()
            os.Args = []string{"apexctl", "status", "--api", srv.URL, "--test", "probe", "--format", tc.format}
            os.Stdout = output
            defer func(){ os.Args, os.Stdout = previousArgs, previousStdout }()
            statusCmd()
            data, err := os.ReadFile(outputPath)
            if err != nil { t.Fatal(err) }
            t.Logf("actual output: %s", data)
            if tc.format == "json" {
                var got map[string]any
                if err := json.Unmarshal(data, &got); err != nil { t.Fatal(err) }
                if evaluated, exists := got["sla_evaluated"]; !exists || evaluated != false {
                    t.Errorf("unevaluated SLA information must survive status JSON: %s", data)
                }
            } else if tc.evaluated {
                if !strings.Contains(string(data), "SLA:    passing") { t.Errorf("evaluated control lost passing verdict: %s", data) }
            } else if strings.Contains(string(data), "SLA:    passing") {
                t.Errorf("unevaluated SLA was reported as passing: %s", data)
            }
        })
    }
}

func TestFullReviewExplainAllActionScopes(t *testing.T) {
    signal := `{"name":"create","type":"request","method":"POST","target":"/orders","extractors":[{"name":"id","source":"body","type":"json_path","expression":"$.id"}],"assertions":[{"type":"status_code","expected":201}]}`
    plain := `{"name":"read","type":"request","method":"GET","target":"/orders"}`
    for _, tc := range []struct{name, sections string; actions int}{
        {"top_level_control", `"actions":[`+signal+`]`, 1},
        {"nested_loop", `"actions":[{"name":"loop","type":"control_loop","control":{"count":1,"steps":[`+signal+`]}}]`, 2},
        {"init_phase", `"init":[`+signal+`],"actions":[`+plain+`]`, 2},
        {"teardown_phase", `"actions":[`+plain+`],"teardown":[`+signal+`]`, 2},
    } {
        t.Run(tc.name, func(t *testing.T) {
            path := writeTempScript(t, `{"name":"explain-probe","protocol":"http","config":{"base_url":"https://example.com"},`+tc.sections+`}`)
            result, err := explainScriptFile(path)
            if err != nil { t.Fatalf("valid probe fixture failed parsing/validation: %v", err) }
            t.Logf("actual result: %#v", result)
            if result.Assertions != 1 || result.Extractors != 1 {
                t.Errorf("lost known assertion/extractor from valid script: assertions=%d extractors=%d", result.Assertions, result.Extractors)
            }
            for _, recommendation := range result.NextActions {
                if strings.HasPrefix(recommendation, "Add assertions") || strings.HasPrefix(recommendation, "Review responses") {
                    t.Errorf("misleading missing-signal recommendation: %s", recommendation)
                }
            }
        })
    }
}
