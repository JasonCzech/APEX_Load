package main
import (
 "context"
 "encoding/json"
 "net/http"
 "net/http/httptest"
 "path/filepath"
 "sync/atomic"
 "testing"
 "time"
)
func TestFullReviewCICancelDuringPoll(t *testing.T) {
 withCIRetryPolicy(t,1,time.Millisecond,time.Millisecond,time.Second,time.Second)
 oldPoll:=ciPollInterval;ciPollInterval=time.Millisecond;t.Cleanup(func(){ciPollInterval=oldPoll})
 for _,tc:=range []struct{name string;cancelPoll bool}{{"cancel during poll",true},{"ordinary transport error control",false}} {
  t.Run(tc.name,func(t *testing.T){
   ctx,cancel:=context.WithCancel(context.Background());defer cancel()
   pollStarted:=make(chan struct{});var aborts atomic.Int32
   server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
    w.Header().Set("Content-Type","application/json")
    switch {
    case r.Method==http.MethodPost && r.URL.Path=="/api/v1/ci/runs":
     w.WriteHeader(http.StatusAccepted);_ = json.NewEncoder(w).Encode(CIRunResponse{SchemaVersion:1,TestID:"test-1",Status:"RUNNING"})
    case r.Method==http.MethodGet && r.URL.Path=="/api/v1/ci/runs/test-1":
     if tc.cancelPoll {close(pollStarted);<-r.Context().Done();return};http.Error(w,"temporary error",500)
    case r.Method==http.MethodPost && r.URL.Path=="/api/v1/ci/runs/test-1/abort":
     aborts.Add(1);_ = json.NewEncoder(w).Encode(CIRunResponse{SchemaVersion:1,TestID:"test-1",Status:"ABORTED"})
    default:http.NotFound(w,r)
    }
   }));defer server.Close()
   if tc.cancelPoll {go func(){select{case <-pollStarted:cancel();case <-ctx.Done():}}()}
   args:=defaultCIArgs();args.controllerURL=server.URL;args.projectID="project-1";args.scriptID="script-1";args.scriptPath="";args.authMode="api-key";args.apiKey="key";args.abortOnCancel=true
   dir:=t.TempDir();args.jsonPath=filepath.Join(dir,"result.json");args.junitPath=filepath.Join(dir,"result.xml")
   result,err:=executeControllerCI(ctx,args,nil)
   if tc.cancelPoll {
    if aborts.Load()!=1 || result.Status!="CANCELLED" || result.Outcome!=ciOutcomeCancelled {t.Fatalf("cancelled poll: aborts=%d status=%s outcome=%s err=%v",aborts.Load(),result.Status,result.Outcome,err)}
   } else if err==nil || aborts.Load()!=0 || result.Status!="STATUS_FAILED" || result.Outcome!=ciOutcomeExecutionError {t.Fatalf("transport control: aborts=%d result=%+v err=%v",aborts.Load(),result,err)}
  })
 }
}
