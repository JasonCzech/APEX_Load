package api

import (
	"bytes"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"io"
	"mime/multipart"
	"net/http"
	"net/http/httptest"
	"os"
	"testing"
	"time"

	"github.com/jasonczech/apexload/pkg/convert/sample"
	"github.com/jasonczech/apexload/pkg/dsl"
	"github.com/jasonczech/apexload/pkg/importartifact"
	"github.com/jasonczech/apexload/pkg/secretstore"
)

// The external archive is never checked in, replayed, or printed. This opt-in
// exercises the same multipart ZIP upload and review conversion as the UI.
func TestExternalHARImportCompatibility(t *testing.T) {
	path := os.Getenv("APEXLOAD_HAR_COMPAT_FILE")
	if path == "" {
		t.Skip("set APEXLOAD_HAR_COMPAT_FILE to a privately held capture")
	}
	source, err := os.Open(path)
	if err != nil {
		t.Fatal("cannot open private capture")
	}
	defer source.Close()
	hash := sha256.New()
	size, err := io.Copy(hash, source)
	if err != nil {
		t.Fatal("cannot hash private capture")
	}
	original := hex.EncodeToString(hash.Sum(nil))
	if want := os.Getenv("APEXLOAD_HAR_COMPAT_SHA256"); want != "" && want != original {
		t.Fatal("private capture hash mismatch")
	}
	if _, err = source.Seek(0, io.SeekStart); err != nil {
		t.Fatal("cannot rewind private capture")
	}
	server, records, _, project := newProjectSecretAPIServer(t)
	closeBatchTestServer(t, server)
	wrapper, err := importartifact.NewLocalAESKeyWrapper(importartifact.LocalKeyOptions{Key: bytes.Repeat([]byte{0x44}, 32)})
	if err != nil {
		t.Fatal("cannot initialize disposable source protection")
	}
	server.protectedSourceSessions = &secretstore.ProtectedSourceSessionStore{Records: records, Wrapper: wrapper}
	endpoint := httptest.NewServer(server.httpServer.Handler)
	defer endpoint.Close()
	prefix := &bytes.Buffer{}
	writer := multipart.NewWriter(prefix)
	for k, v := range map[string]string{"project_id": project, "name": "External HAR compatibility"} {
		if err := writer.WriteField(k, v); err != nil {
			t.Fatal("multipart setup failed")
		}
	}
	if _, err = writer.CreateFormFile("file", "external-capture.zip"); err != nil {
		t.Fatal("multipart setup failed")
	}
	suffix := "\r\n--" + writer.Boundary() + "--\r\n"
	req, err := http.NewRequest(http.MethodPost, endpoint.URL+"/api/v1/scripts/import/har?project_id="+project, io.MultiReader(bytes.NewReader(prefix.Bytes()), source, bytes.NewBufferString(suffix)))
	if err != nil {
		t.Fatal("request setup failed")
	}
	req.ContentLength = int64(prefix.Len()) + size + int64(len(suffix))
	req.Header.Set("Content-Type", writer.FormDataContentType())
	client := endpoint.Client()
	client.Timeout = 5 * time.Minute
	response, err := client.Do(req)
	if err != nil {
		t.Fatal("private HAR HTTP request failed")
	}
	defer response.Body.Close()
	if response.StatusCode != http.StatusOK {
		t.Fatalf("private HAR import status=%d; response withheld", response.StatusCode)
	}
	var result struct {
		Script        dsl.Script          `json:"script"`
		Stats         sample.ConvertStats `json:"stats"`
		Warnings      []string            `json:"warnings"`
		SourceSession string              `json:"source_secret_session_id"`
		Requirements  []json.RawMessage   `json:"secret_binding_requirements"`
	}
	if err := json.NewDecoder(io.LimitReader(response.Body, 300<<20)).Decode(&result); err != nil {
		t.Fatal("private HAR response decoding failed")
	}
	if result.Stats.TotalRequests != 803 || result.Stats.ConvertedActions < 1 || len(result.Script.Actions) == 0 {
		t.Fatal("private HAR request population was not converted")
	}
	if result.Script.ProjectID != project {
		t.Fatal("import lost project scope")
	}
	validation := dsl.ValidateScriptIssues(&result.Script)
	result.Script.ID = ""
	result.Script.CreatedAt = time.Time{}
	canonical, err := json.Marshal(result.Script)
	if err != nil {
		t.Fatal("private script canonicalization failed")
	}
	if privateOutput := os.Getenv("APEXLOAD_HAR_COMPAT_PRIVATE_SCRIPT"); privateOutput != "" {
		if err := os.WriteFile(privateOutput, canonical, 0600); err != nil {
			t.Fatal("private comparison output failed")
		}
	}
	scriptHash := sha256.Sum256(canonical)
	shape := map[string]int{}
	for _, action := range result.Script.Actions {
		shape[action.Type]++
	}
	summary := map[string]any{"archive_sha256": original, "archive_bytes": size, "status": response.StatusCode, "stats": result.Stats, "actions": len(result.Script.Actions), "top_level_action_types": shape, "warning_count": len(result.Warnings), "validation_issue_count": len(validation), "protected_requirements": len(result.Requirements), "protected_source_session": result.SourceSession != "", "script_sha256_without_id_created_at": hex.EncodeToString(scriptHash[:])}
	if _, err = source.Seek(0, io.SeekStart); err != nil {
		t.Fatal("cannot verify original capture")
	}
	hash.Reset()
	if _, err = io.Copy(hash, source); err != nil || hex.EncodeToString(hash.Sum(nil)) != original {
		t.Fatal("original capture changed")
	}
	if output := os.Getenv("APEXLOAD_HAR_COMPAT_SUMMARY"); output != "" {
		data, err := json.MarshalIndent(summary, "", "  ")
		if err != nil {
			t.Fatal("summary encoding failed")
		}
		if err = os.WriteFile(output, append(data, '\n'), 0600); err != nil {
			t.Fatal("summary write failed")
		}
	}
	t.Logf("private ZIP imported: %d source requests, %d converted actions, %d top-level actions; original hash unchanged", result.Stats.TotalRequests, result.Stats.ConvertedActions, len(result.Script.Actions))
}
