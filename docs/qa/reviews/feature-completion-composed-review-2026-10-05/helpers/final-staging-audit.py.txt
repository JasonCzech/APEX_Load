"""Pure prospective staging audit. Never stages, changes source/index, or cleans up.

Run only after physical joins, then independently review audit.json and complete.json.
Successful output includes exact relative-file NUL plans: all, normal, and force.
Any subsequent staging is a separate human/parent operation; use literal pathspecs.
New owned-prefix QA is discovery, not a claim of earlier review or qualification.
Outputs are create-only /private/tmp files, mode 0444, with a completion hash manifest;
this is write-once audit discipline, not an OS immutable flag or a provenance review.
"""

def sha256(data):
    return hashlib.sha256(data).hexdigest()


def git(*args, input_bytes=None, allowed_codes=(0,)):
    # Disallow optional index refresh/locks; commands below are read-only getters.
    env = os.environ.copy()
    env['GIT_OPTIONAL_LOCKS'] = '0'
    p = subprocess.run(['git', '--no-optional-locks', '-C', str(ROOT), *args],
                       input=input_bytes, capture_output=True, env=env)
    if p.returncode not in allowed_codes:
        raise RuntimeError('read-only git command failed: ' + ' '.join(args[:2]) +
                           ' (exit ' + str(p.returncode) + ')')
    return p.stdout


def decode_path(raw):
    path = raw.decode('utf-8', errors='strict')
    parts = PurePosixPath(path).parts
    if (not path or path.startswith('/') or '\\' in path or '\0' in path or
            any(part in ('.', '..', '.git') for part in parts) or
            PurePosixPath(path).as_posix() != path):
        raise RuntimeError('invalid repository-relative path in metadata')
    return path


def nul_paths(raw):
    return [decode_path(p) for p in raw.split(b'\0') if p]


def read_status():
    raw = git('status', '--porcelain=v1', '-z', '--untracked-files=all')
    chunks = raw.split(b'\0')
    result = {}
    i = 0
    while i < len(chunks):
        c = chunks[i]
        i += 1
        if not c:
            continue
        if len(c) < 4 or c[2:3] != b' ':
            raise RuntimeError('unexpected git status encoding')
        status = c[:2].decode('ascii')
        path = decode_path(c[3:])
        if path in result:
            raise RuntimeError('duplicate git status path')
        result[path] = {'git_status': status}
        if 'R' in status or 'C' in status:
            if i >= len(chunks) or not chunks[i]:
                raise RuntimeError('missing rename/copy source path')
            origin = decode_path(chunks[i])
            i += 1
            result[path]['rename_or_copy_origin'] = origin
            if 'R' in status:
                result.setdefault(origin, {'git_status': ' D',
                                           'rename_origin_for': path})
    return raw, result


def state():
    head = git('rev-parse', 'HEAD').decode().strip()
    branch = git('branch', '--show-current').decode().strip()
    index_name = git('rev-parse', '--git-path', 'index').decode().strip()
    index_path = Path(index_name)
    if not index_path.is_absolute():
        index_path = ROOT / index_path
    return {'head': head, 'branch': branch,
            'index_sha256': sha256(index_path.read_bytes()) if index_path.is_file() else None,
            'staged_paths': nul_paths(git('diff', '--cached', '--name-only', '-z'))}


def excluded_reason(path):
    if path in EXPLICIT_EXCLUSIONS:
        return 'explicit required exclusion'
    parts = PurePosixPath(path).parts
    if any(p.startswith('.playwright-artifacts-') for p in parts):
        return 'browser runner scratch; never a deliverable'
    if any(p in ('node_modules', '__pycache__', '.vite', '.vite-temp', '.cache', '.pytest_cache', '.git') for p in parts):
        return 'dependency/cache/private repository metadata'
    if any(path == p.rstrip('/') or path.startswith(p) for p in EXCLUDED_DIR_PREFIXES):
        return 'dependency/cache/application artifact output'
    base = parts[-1] if parts else ''
    if base.lower() == 'wls_multilinecrp.har' or base.lower().endswith(('.har', '.har.gz')):
        return 'raw HAR/private capture exclusion; content is never opened'
    return None


def scope_of(path, sources):
    if path in QA_STANDALONE:
        return 'explicit reviewed QA document/generated deliverable'
    if path.startswith(QA_PREFIXES):
        return 'owned retained QA prefix'
    if path.startswith(STATIC_PREFIX):
        return 'generated embedded dashboard deliverable'
    if path in sources:
        return 'explicit reviewed non-QA source/document candidate'
    return None


def identity(st):
    return (st.st_dev, st.st_ino, st.st_mode, st.st_size, st.st_mtime_ns, st.st_ctime_ns)


def ensure_no_symlink_components(path):
    current = ROOT
    for part in PurePosixPath(path).parts:
        current = current / part
        try:
            st = current.lstat()
        except FileNotFoundError:
            return
        if stat.S_ISLNK(st.st_mode):
            raise RuntimeError('symlink path/component is excluded')


def read_regular_hash(path):
    ensure_no_symlink_components(path)
    full = ROOT / path
    before = full.lstat()
    if not stat.S_ISREG(before.st_mode):
        raise RuntimeError('candidate is not a regular file')
    if before.st_size >= MAX_FILE_BYTES:
        raise RuntimeError('candidate is at least 100 MiB')
    fd = os.open(full, os.O_RDONLY | getattr(os, 'O_NOFOLLOW', 0))
    try:
        opened = os.fstat(fd)
        if identity(opened) != identity(before):
            raise RuntimeError('candidate changed during open')
        h = hashlib.sha256()
        total = 0
        while True:
            data = os.read(fd, 1024 * 1024)
            if not data:
                break
            total += len(data)
            if total >= MAX_FILE_BYTES:
                raise RuntimeError('candidate grew to at least 100 MiB')
            h.update(data)
        after_fd = os.fstat(fd)
    finally:
        os.close(fd)
    after_path = full.lstat()
    if identity(before) != identity(after_fd) or identity(before) != identity(after_path):
        raise RuntimeError('candidate changed during hash')
    return {'bytes': before.st_size, 'sha256': h.hexdigest(),
            'mode': stat.S_IMODE(before.st_mode), 'device': before.st_dev,
            'inode': before.st_ino, 'mtime_ns': before.st_mtime_ns,
            'ctime_ns': before.st_ctime_ns}, identity(before)


def discover_scoped_files(excluded, issues):
    found = set()
    for prefix in (*QA_PREFIXES, STATIC_PREFIX):
        directory = ROOT / prefix.rstrip('/')
        if not directory.exists() and not directory.is_symlink():
            continue
        if directory.is_symlink():
            excluded[prefix.rstrip('/')] = 'symlink owned scope; never followed'
            issues.append({'path': prefix.rstrip('/'), 'reason': 'owned scope is a symlink'})
            continue
        for parent, dirs, files in os.walk(directory, followlinks=False):
            for name in list(dirs):
                child = Path(parent) / name
                rel = child.relative_to(ROOT).as_posix()
                reason = excluded_reason(rel)
                if child.is_symlink():
                    reason = 'symlink directory; never followed'
                    issues.append({'path': rel, 'reason': 'symlink inside owned scope'})
                if reason:
                    dirs.remove(name)
                    excluded[rel] = reason
                    if name.startswith('.playwright-artifacts-'):
                        issues.append({'path': rel, 'reason': 'residual runner scratch prevents final passed snapshot'})
            for name in files:
                child = Path(parent) / name
                rel = child.relative_to(ROOT).as_posix()
                reason = excluded_reason(rel)
                if reason:
                    excluded[rel] = reason
                    continue
                if child.is_symlink():
                    excluded[rel] = 'symlink file; never followed'
                    issues.append({'path': rel, 'reason': 'symlink inside owned scope'})
                    continue
                found.add(rel)
    return found


def ignored_rules(paths):
    if not paths:
        return {}
    raw = git('check-ignore', '-v', '-z', '--stdin',
              input_bytes=b''.join(p.encode('utf-8') + b'\0' for p in sorted(paths)),
              allowed_codes=(0, 1))
    chunks = [p for p in raw.split(b'\0')]
    if chunks and chunks[-1] == b'':
        chunks.pop()
    if len(chunks) % 4:
        raise RuntimeError('unexpected check-ignore encoding')
    result = {}
    for i in range(0, len(chunks), 4):
        source, line, pattern, path = chunks[i:i + 4]
        # Verbose check-ignore can report a negating rule; that path is not ignored.
        if pattern.startswith(b'!'):
            continue
        result[decode_path(path)] = {'source': source.decode('utf-8'),
                                    'line': int(line), 'pattern': pattern.decode('utf-8')}
    return result


def create_readonly_file(directory, name, data):
    fd = os.open(directory / name, os.O_WRONLY | os.O_CREAT | os.O_EXCL |
                 getattr(os, 'O_NOFOLLOW', 0), 0o600)
    try:
        view = memoryview(data)
        while view:
            count = os.write(fd, view)
            if count <= 0:
                raise RuntimeError('audit output write did not progress')
            view = view[count:]
        os.fsync(fd)
        os.fchmod(fd, 0o444)
    finally:
        os.close(fd)
    return {'path': name, 'bytes': len(data), 'sha256': sha256(data)}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--gates-physically-joined', action='store_true', required=True,
                        help='caller attests all writers/builds/browser runners have physically joined')
    parser.add_argument('--output-name', required=True,
                        help='fresh direct-child directory name under /private/tmp (never reused)')
    args = parser.parse_args()
    if not args.gates_physically_joined:
        parser.error('physical gate join attestation is required')
    if (not args.output_name or args.output_name in ('.', '..') or
            '/' in args.output_name or '\\' in args.output_name or '\0' in args.output_name):
        parser.error('output-name must be one fresh /private/tmp directory basename')
    output = Path('/private/tmp') / args.output_name
    if output.exists() or output.is_symlink():
        parser.error('create-only output directory already exists')
    raw_advisory = ADVISORY.read_bytes()
    if sha256(raw_advisory) != ADVISORY_SHA256:
        raise RuntimeError('advisory hash mismatch')
    advisory = json.loads(raw_advisory)
    if (advisory['root'] != str(ROOT) or advisory['head'] != EXPECTED_HEAD or
            tuple(advisory['owned_qa_allowlist']) != QA_PREFIXES or
            set(advisory['explicit_exclusions']) != EXPLICIT_EXCLUSIONS):
        raise RuntimeError('advisory root/HEAD/scope mismatch')
    if not ROOT.is_dir() or ROOT.is_symlink():
        raise RuntimeError('expected ROOT directory unavailable or symlinked')
    before_state = state()
    if before_state['head'] != EXPECTED_HEAD or before_state['branch'] != EXPECTED_BRANCH:
        raise RuntimeError('ROOT HEAD/branch mismatch; do not adopt changed branch automatically')
    if before_state['staged_paths']:
        raise RuntimeError('pre-stage index is not empty; refuse combining unseen staged changes')
    helper_sha = sha256(Path(__file__).read_bytes())
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    before_status_raw, statuses = read_status()
    old_rows = {r['path']: r for r in advisory['rows']}
    sources = {r['path'] for r in advisory['rows']
               if r['action'].startswith('candidate') and
               not r['path'].startswith(('docs/qa/', STATIC_PREFIX))}
    sources.update(QA_STANDALONE)
    tracked = set(nul_paths(git('ls-files', '-z')))
    excluded = {p: 'explicit required exclusion' for p in EXPLICIT_EXCLUSIONS}
    issues = []
    selected = set(sources)
    selected.update(p for p in tracked if scope_of(p, sources))
    selected.update(discover_scoped_files(excluded, issues))
    for path in statuses:
        reason = excluded_reason(path)
        if reason:
            excluded[path] = reason
        elif scope_of(path, sources):
            selected.add(path)
        else:
            issues.append({'path': path, 'reason': 'unknown dirty/untracked source or unowned QA path'})
    selected = {p for p in selected if not excluded_reason(p)}
    rules = ignored_rules(selected - tracked)
    rows = []
    first_identities = {}
    for path in sorted(selected, key=lambda p: p.encode('utf-8')):
        row = {'path': path, 'scope': scope_of(path, sources),
               'git_status': statuses.get(path, {}).get('git_status', '  '),
               'tracked': path in tracked, 'ignored': path in rules,
               'ignore_rule': rules.get(path), 'in_prior_advisory': path in old_rows,
               'scope_review_claim': ('Path previously present in advisory; final bytes still require parent review.'
                                      if path in old_rows else
                                      'New retained deliverable in owned scope; no prior review/qualification claim.')}
        row.update({k: v for k, v in statuses.get(path, {}).items() if k != 'git_status'})
        full = ROOT / path
        if not full.exists() and not full.is_symlink():
            row['file_type'] = 'absent'
            row['bytes'] = 0
            row['sha256'] = None
            if path not in tracked or path not in statuses:
                issues.append({'path': path, 'reason': 'reviewed candidate disappeared without tracked deletion'})
            row['stage_action'] = 'normal' if path in tracked and path in statuses else 'none'
            rows.append(row)
            continue
        try:
            # Refused files retain size/type metadata, without reading their content.
            ensure_no_symlink_components(path)
            candidate_stat = full.lstat()
            row.update({'bytes': candidate_stat.st_size,
                        'mode': stat.S_IMODE(candidate_stat.st_mode),
                        'mtime_ns': candidate_stat.st_mtime_ns,
                        'file_type': 'regular' if stat.S_ISREG(candidate_stat.st_mode) else 'non-regular'})
            info, ident = read_regular_hash(path)
            row.update(info)
            row['file_type'] = 'regular'
            first_identities[path] = ident
            expected = old_rows.get(path, {}).get('sha256_at_ownership_mapping')
            if expected and row['sha256'] != expected:
                issues.append({'path': path, 'reason': 'parent-confirmed resource mapping hash changed'})
        except (OSError, RuntimeError) as e:
            row['file_type'] = 'unqualified; not staged'
            row['error'] = type(e).__name__ + ': ' + str(e)
            issues.append({'path': path, 'reason': row['error']})
        dirty = path in statuses or path not in tracked
        row['stage_action'] = ('force' if path in rules else 'normal') if dirty else 'none'
        rows.append(row)
    # Bookend every selected regular file with identity and logical SHA, not just status.
    row_by_path = {r['path']: r for r in rows}
    for path, original_identity in first_identities.items():
        try:
            info, final_identity = read_regular_hash(path)
            if (final_identity != original_identity or
                    info['sha256'] != row_by_path[path]['sha256']):
                issues.append({'path': path, 'reason': 'candidate changed between audit hash bookends'})
        except (OSError, RuntimeError) as e:
            issues.append({'path': path, 'reason': 'final hash bookend failed: ' + type(e).__name__})
    after_status_raw, _ = read_status()
    after_state = state()
    if before_status_raw != after_status_raw:
        issues.append({'reason': 'ROOT git status changed during audit'})
    if before_state != after_state:
        issues.append({'reason': 'ROOT HEAD/branch/index changed during audit'})
    # Re-discover to catch ignored additions/deletions absent from ordinary git status.
    final_scoped = discover_scoped_files(excluded, issues)
    initial_scoped = {p for p in selected if p.startswith((*QA_PREFIXES, STATIC_PREFIX))
                      and p in first_identities}
    if final_scoped != initial_scoped:
        issues.append({'reason': 'owned QA/static filesystem membership changed during audit',
                       'added_paths': sorted(final_scoped - initial_scoped),
                       'removed_paths': sorted(initial_scoped - final_scoped)})
    # Do not publish any usable pathspec on a refused/unstable audit.
    if sha256(Path(__file__).read_bytes()) != helper_sha:
        issues.append({'reason': 'audit helper changed during execution'})
    passed = not issues
    plan = sorted((r['path'] for r in rows if r['stage_action'] != 'none'),
                  key=lambda p: p.encode('utf-8')) if passed else []
    normal = [p for p in plan if row_by_path[p]['stage_action'] == 'normal']
    force = [p for p in plan if row_by_path[p]['stage_action'] == 'force']
    audit = {'schema': 'apexload-root-final-prospective-staging-audit-v1',
             'scope': 'Pure read-only prospective audit; no git mutation, staging, cleanup, source write, target traffic or private raw capture/trace-payload parsing. New QA discovery is not a review or acceptance claim.',
             'root': str(ROOT), 'helper_sha256': helper_sha,
             'expected_head': EXPECTED_HEAD, 'expected_branch': EXPECTED_BRANCH,
             'caller_attests_physical_gate_joins': True, 'started_utc': started,
             'finished_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
             'advisory': {'path': str(ADVISORY), 'sha256': ADVISORY_SHA256},
             'head_branch_index_before': before_state, 'head_branch_index_after': after_state,
             'status_before_sha256': sha256(before_status_raw),
             'status_after_sha256': sha256(after_status_raw),
             'owned_qa_prefixes': list(QA_PREFIXES), 'explicit_qa_deliverables': sorted(QA_STANDALONE),
             'generated_static_prefix': STATIC_PREFIX, 'max_file_bytes_exclusive': MAX_FILE_BYTES,
             'passed': passed, 'issues': issues, 'rows': rows,
             'excluded': [{'path': p, 'reason': excluded[p]} for p in sorted(excluded)],
             'summary': {'rows': len(rows), 'known_non_QA_candidates': len(sources - QA_STANDALONE),
                         'logical_regular_bytes': sum(r.get('bytes', 0) for r in rows),
                         'normal_stage_paths': len(normal), 'force_stage_paths': len(force),
                         'all_stage_paths': len(plan), 'new_owned_scope_deliverables':
                         sum(not r['in_prior_advisory'] for r in rows),
                         'scope_counts': dict(collections.Counter(r['scope'] for r in rows)),
                         'large_files_10_MiB_or_more': [r['path'] for r in rows
                                                      if r.get('bytes', 0) >= 10 * 1024 * 1024]},
             'separate_staging_contract': 'Parent must review this exact completed audit and provenance, then stage separately using only normal-paths.nul and explicit force-paths.nul, with --literal-pathspecs and --pathspec-file-nul. Never git add -A. Reaudit if any path/HEAD/index/output changes. Excluded paths and directories are never staging candidates.'}
    os.mkdir(output, 0o700)  # Exclusive, create-only, only under /private/tmp.
    results = []
    results.append(create_readonly_file(output, 'audit.json',
                   (json.dumps(audit, indent=2, ensure_ascii=False) + '\n').encode('utf-8')))
    if passed:
        for name, paths in (('all-paths.nul', plan), ('normal-paths.nul', normal),
                            ('force-paths.nul', force)):
            results.append(create_readonly_file(output, name,
                           b''.join(p.encode('utf-8') + b'\0' for p in paths)))
    complete = {'schema': 'apexload-root-audit-output-completion-v1', 'passed': passed,
                'files': results, 'pathspecs_created': passed,
                'output_permissions': 'create-only files 0444; directory 0500 after completion',
                'logical_immutability_limit': 'Owner/root can change permissions; no filesystem immutable flag set.'}
    create_readonly_file(output, 'complete.json',
                         (json.dumps(complete, indent=2) + '\n').encode('utf-8'))
    dfd = os.open(output, os.O_RDONLY)
    try:
        os.fsync(dfd)
    finally:
        os.close(dfd)
    os.chmod(output, 0o500)
    print(json.dumps({'output_directory': str(output), 'passed': passed,
                      'audit_sha256': results[0]['sha256'], 'issues': len(issues),
                      'normal_paths': len(normal), 'force_paths': len(force),
                      'pathspecs_created': passed}, indent=2))
    return 0 if passed else 2


if __name__ == '__main__':
    try:
        sys.exit(main())
    except (OSError, RuntimeError, ValueError) as exc:
        # Controlled diagnostics only; no subprocess stderr/source/content output.
        print('Audit refused: ' + str(exc), file=sys.stderr)
        sys.exit(2)
