package playwright

import (
	"errors"
	"sync"
)

// ErrHARCloseOwnershipUnproven means no fresh, fully completed HAR writer close
// can be certified. A prior ordinary/server close is never adopted as proof.
var ErrHARCloseOwnershipUnproven = errors.New("HAR writer close ownership unproven")

// OwnedHARBrowserContext is an optional APEX extension. Ordinary Close keeps
// its upstream early-nil compatibility; this method returns a joined result.
type OwnedHARBrowserContext interface {
	CloseHAROwned(...BrowserContextCloseOptions) error
}

type ownedHARCloseState struct {
	mu     sync.Mutex
	done   chan struct{}
	result error
}

func (b *browserContextImpl) CloseHAROwned(options ...BrowserContextCloseOptions) error {
	return b.closeHAROwned(func() error { return b.closeEntered(options...) })
}

// closeHAROwned publishes only the complete physical body's result. No mutex
// is held during native work or while repeat callers await its receipt.
func (b *browserContextImpl) closeHAROwned(work func() error) (err error) {
	state := &b.ownedHARClose
	state.mu.Lock()
	if state.done != nil {
		done := state.done
		state.mu.Unlock()
		<-done
		state.mu.Lock()
		result := state.result
		state.mu.Unlock()
		return result
	}
	if b.isClosedFlag.Load() || !b.closeWasCalled.CompareAndSwap(false, true) {
		state.mu.Unlock()
		return ErrHARCloseOwnershipUnproven
	}
	state.done = make(chan struct{})
	done := state.done
	state.mu.Unlock()
	completed := false
	defer func() {
		recovered := recover()
		if !completed || recovered != nil || err != nil {
			err = errors.Join(err, ErrHARCloseOwnershipUnproven)
		}
		state.mu.Lock()
		state.result = err
		close(done)
		state.mu.Unlock()
		if recovered != nil {
			panic(recovered)
		}
	}()
	err = work()
	completed = true
	return err
}
